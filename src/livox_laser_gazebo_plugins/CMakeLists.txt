cmake_minimum_required(VERSION 3.0.2)
project(livox_laser_gazebo_plugins)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs pcl_conversions)
find_package(gazebo REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
catkin_package(INCLUDE_DIRS include LIBRARIES livox_laser_gazebo_plugins CATKIN_DEPENDS roscpp sensor_msgs pcl_conversions)
include_directories(include ${catkin_INCLUDE_DIRS} ${GAZEBO_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
add_library(livox_laser_gazebo_plugins SHARED src/livox_points_plugin.cpp src/livox_ode_multiray_shape.cpp)
target_link_libraries(livox_laser_gazebo_plugins ${catkin_LIBRARIES} ${GAZEBO_LIBRARIES} RayPlugin)
install(TARGETS livox_laser_gazebo_plugins LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
