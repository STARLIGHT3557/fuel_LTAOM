cmake_minimum_required(VERSION 3.0.2)
project(prometheus_fuel)
add_compile_options(-std=c++14 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp nav_msgs sensor_msgs geometry_msgs
  std_msgs std_srvs visualization_msgs message_filters pcl_ros pcl_conversions
  uav_controller uav_mission_msgs quadrotor_msgs active_perception)
find_package(Eigen3 REQUIRED)
catkin_package()
include_directories(${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
add_executable(fuel_bridge src/fuel_bridge.cpp)
add_dependencies(fuel_bridge ${catkin_EXPORTED_TARGETS})
target_link_libraries(fuel_bridge ${catkin_LIBRARIES})
add_executable(mission_manager_node src/mission_manager_node.cpp)
add_dependencies(mission_manager_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(mission_manager_node ${catkin_LIBRARIES})
install(TARGETS fuel_bridge mission_manager_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(PROGRAMS scripts/fuel_env.sh DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  add_rostest(test/bridge.test)
  add_rostest(test/planner.test)
  add_rostest(test/planner_overrun.test)
  add_rostest(test/exploration_handover.test)
  add_rostest(test/exploration_timeouts.test)
  add_rostest(test/return_recovery.test)
  add_rostest(test/return_planner.test)
  add_rostest_gtest(fuel_frontier_boundary test/frontier_boundary.test test/test_frontier_boundary.cpp)
  target_include_directories(fuel_frontier_boundary PRIVATE ${active_perception_INCLUDE_DIRS})
  target_link_libraries(fuel_frontier_boundary ${active_perception_LIBRARIES} ${catkin_LIBRARIES})
endif()
