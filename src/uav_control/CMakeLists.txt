cmake_minimum_required(VERSION 3.0.2)
project(prometheus_uav_control)
add_compile_options(-std=c++14 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs nav_msgs sensor_msgs std_msgs mavros_msgs prometheus_msgs visualization_msgs tf2 tf2_ros)
find_package(Eigen3 REQUIRED)
catkin_package()
include_directories(include ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
add_executable(uav_control_main src/uav_control_node.cpp src/uav_controller.cpp src/uav_estimator.cpp)
add_dependencies(uav_control_main ${catkin_EXPORTED_TARGETS})
target_link_libraries(uav_control_main ${catkin_LIBRARIES})
install(TARGETS uav_control_main RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config meshes DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  add_rostest(test/controller.test)
endif()
